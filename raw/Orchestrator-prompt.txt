```typescript
// components/book/BookCard.tsx

import React from 'react';
import { Pressable, View, Text } from 'react-native';
import { Image } from 'expo-image';
import Animated, { useSharedValue, useAnimatedStyle, withSpring } from 'react-native-reanimated';
import { Icon } from '@/components/ui/Icon';
import { colors, spacing, radius, shadows, typography } from '@/constants';
import type { Book } from '@/types';

// ─── Types ───────────────────────────────────────────────────────────────────
interface BookCardProps {
  book: Book;
  onPress: (slug: string) => void;
  variant?: 'grid' | 'list' | 'featured';
  isCompleted?: boolean;
}

// ─── Component ───────────────────────────────────────────────────────────────
export function BookCard({ book, onPress, variant = 'grid', isCompleted = false }: BookCardProps) {
  const scale = useSharedValue(1);

  const animatedStyle = useAnimatedStyle(() => ({
    transform: [{ scale: scale.value }],
  }));

  const handlePressIn = () => { scale.value = withSpring(0.97, { damping: 15 }); };
  const handlePressOut = () => { scale.value = withSpring(1, { damping: 15 }); };

  return (
    <Pressable onPress={() => onPress(book.slug)} onPressIn={handlePressIn} onPressOut={handlePressOut}>
      <Animated.View style={[animatedStyle, shadows.md, { borderRadius: radius.lg, overflow: 'hidden' }]}>
        {/* content */}
      </Animated.View>
    </Pressable>
  );
}
```

**Rules:**
- Every interactive element has a press animation (spring scale 0.97)
- Never use `TouchableOpacity` — always `Pressable` + Reanimated
- Every component exports a named export (not default)
- Props interface always defined above the component
- Organize with `─── Section ───` comments

## STATE MANAGEMENT RULES

### Zustand Store Pattern

```typescript
// store/userStore.ts
import { create } from 'zustand';
import { persist, createJSONStorage } from 'zustand/middleware';
import { zustandMMKVStorage } from '@/lib/mmkv';

interface UserState {
  // State
  deviceId: string | null;
  streak: number;
  lastActiveDate: string | null;
  booksCompleted: string[];
  booksInProgress: string[];
  theme: 'light' | 'dark' | 'sepia';

  // Actions — always co-located with state
  setDeviceId: (id: string) => void;
  incrementStreak: () => void;
  markBookCompleted: (bookId: string) => void;
  setTheme: (theme: 'light' | 'dark' | 'sepia') => void;
}

export const useUserStore = create<UserState>()(
  persist(
    (set, get) => ({
      deviceId: null,
      streak: 0,
      lastActiveDate: null,
      booksCompleted: [],
      booksInProgress: [],
      theme: 'light',

      setDeviceId: (id) => set({ deviceId: id }),
      incrementStreak: () => set((s) => ({ streak: s.streak + 1 })),
      markBookCompleted: (bookId) => set((s) => ({
        booksCompleted: [...new Set([...s.booksCompleted, bookId])],
        booksInProgress: s.booksInProgress.filter((id) => id !== bookId),
      })),
      setTheme: (theme) => set({ theme }),
    }),
    {
      name: 'user-store',
      storage: createJSONStorage(() => zustandMMKVStorage),
    }
  )
);
```

**Rules:**
- One store per domain: `userStore`, `bookStore`, `readerStore`, `flashcardStore`
- All stores persisted with MMKV via `zustand/middleware persist`
- Actions always co-located with state (not in separate files)
- Never mutate state directly — always use `set()`
- Selectors for derived state to prevent unnecessary re-renders:

```typescript
// Selector — use this pattern to avoid re-renders
const streak = useUserStore((s) => s.streak); // NOT: const { streak } = useUserStore()
```

```typescript
// lib/api.ts
const BASE_URL = process.env.EXPO_PUBLIC_API_URL ?? 'http://localhost:3000';

async function fetcher<T>(path: string, options?: RequestInit): Promise<T> {
  const res = await fetch(`${BASE_URL}${path}`, {
    headers: { 'Content-Type': 'application/json' },
    ...options,
  });
  if (!res.ok) {
    const error = await res.json().catch(() => ({}));
    throw new Error(error.message ?? `HTTP ${res.status}`);
  }
  return res.json();
}

export const api = {
  books: {
    list: () => fetcher<BookSummary[]>('/api/books'),
    detail: (slug: string) => fetcher<Book>(`/api/books/${slug}`),
    lessons: (slug: string) => fetcher<LessonSummary[]>(`/api/books/${slug}/lessons`),
    lesson: (slug: string, id: string) => fetcher<Lesson>(`/api/books/${slug}/lessons/${id}`),
    flashcards: (slug: string) => fetcher<Flashcard[]>(`/api/books/${slug}/flashcards`),
  },
  user: {
    init: (deviceId: string) => fetcher<User>('/api/user/init', { method: 'POST', body: JSON.stringify({ deviceId }) }),
    updateProgress: (data: ProgressUpdate) => fetcher('/api/user/progress', { method: 'PATCH', body: JSON.stringify(data) }),
    saveHighlight: (data: HighlightCreate) => fetcher('/api/user/highlights', { method: 'POST', body: JSON.stringify(data) }),
  },
};
```

### Custom Hook Pattern

```typescript
// hooks/useBook.ts
import { useState, useEffect, useCallback } from 'react';
import { api } from '@/lib/api';
import type { Book } from '@/types';

interface UseBookResult {
  book: Book | null;
  isLoading: boolean;
  error: Error | null;
  refetch: () => void;
}

export function useBook(slug: string): UseBookResult {
  const [book, setBook] = useState<Book | null>(null);
  const [isLoading, setIsLoading] = useState(true);
  const [error, setError] = useState<Error | null>(null);

  const fetch = useCallback(async () => {
    try {
      setIsLoading(true);
      setError(null);
      const data = await api.books.detail(slug);
      setBook(data);
    } catch (err) {
      setError(err instanceof Error ? err : new Error('Unknown error'));
    } finally {
      setIsLoading(false);
    }
  }, [slug]);

  useEffect(() => { fetch(); }, [fetch]);

  return { book, isLoading, error, refetch: fetch };
}
```

**Rules:**
- Custom hook for every data-fetching use case
- Always expose `isLoading`, `error`, `refetch`
- Never fetch directly inside components
- Cache book list in Zustand `bookStore` after first fetch

### Flashcard Flip Animation

```typescript
// components/flashcard/FlashCard.tsx
import Animated, {
  useSharedValue,
  useAnimatedStyle,
  withSpring,
  interpolate,
} from 'react-native-reanimated';

export function FlashCard({ card, onSwipe }: FlashCardProps) {
  const rotateY = useSharedValue(0);
  const isFlipped = useSharedValue(false);

  const frontStyle = useAnimatedStyle(() => ({
    transform: [{ rotateY: `${interpolate(rotateY.value, [0, 1], [0, 180])}deg` }],
    backfaceVisibility: 'hidden',
  }));

  const backStyle = useAnimatedStyle(() => ({
    transform: [{ rotateY: `${interpolate(rotateY.value, [0, 1], [180, 360])}deg` }],
    backfaceVisibility: 'hidden',
    position: 'absolute',
    top: 0, left: 0, right: 0, bottom: 0,
  }));

  const flip = () => {
    rotateY.value = withSpring(isFlipped.value ? 0 : 1, { damping: 20, stiffness: 300 });
    isFlipped.value = !isFlipped.value;
  };

  return (
    <Pressable onPress={flip}>
      <View style={{ width: CARD_WIDTH, height: CARD_HEIGHT }}>
        <Animated.View style={frontStyle}>{/* Front */}</Animated.View>
        <Animated.View style={backStyle}>{/* Back */}</Animated.View>
      </View>
    </Pressable>
  );
}
```

### Screen Entry Animation

Every screen should fade + slide in on mount:

```typescript
// hooks/useScreenEntrance.ts
import { useEffect } from 'react';
import { useSharedValue, useAnimatedStyle, withSpring, withTiming } from 'react-native-reanimated';

export function useScreenEntrance(delay = 0) {
  const opacity = useSharedValue(0);
  const translateY = useSharedValue(20);

  useEffect(() => {
    const timer = setTimeout(() => {
      opacity.value = withTiming(1, { duration: 300 });
      translateY.value = withSpring(0, { damping: 20, stiffness: 200 });
    }, delay);
    return () => clearTimeout(timer);
  }, []);

  return useAnimatedStyle(() => ({
    opacity: opacity.value,
    transform: [{ translateY: translateY.value }],
  }));
}
```

## NEXT.JS API RULES

### Route Handler Template

```typescript
// app/api/books/[slug]/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { connectDB } from '@/lib/db';
import { Book } from '@/models/Book';
import { z } from 'zod';

const paramsSchema = z.object({ slug: z.string().min(1) });

export async function GET(
  req: NextRequest,
  { params }: { params: { slug: string } }
) {
  // ─── Validate params ────────────────────────────────────────────────
  const parsed = paramsSchema.safeParse(params);
  if (!parsed.success) {
    return NextResponse.json({ message: 'Invalid slug' }, { status: 400 });
  }

  // ─── DB query ───────────────────────────────────────────────────────
  try {
    await connectDB();
    const book = await Book.findOne({ slug: parsed.data.slug }).lean();
    if (!book) {
      return NextResponse.json({ message: 'Book not found' }, { status: 404 });
    }
    return NextResponse.json(book);
  } catch (err) {
    console.error('[GET /api/books/[slug]]', err);
    return NextResponse.json({ message: 'Internal server error' }, { status: 500 });
  }
}
```

### MongoDB Connection (with connection pooling)

```typescript
// lib/db.ts
import mongoose from 'mongoose';

const MONGODB_URI = process.env.MONGODB_URI!;

if (!MONGODB_URI) throw new Error('MONGODB_URI is not set');

// Persist connection across serverless invocations
let cached = global.mongoose as { conn: typeof mongoose | null; promise: Promise<typeof mongoose> | null };

if (!cached) {
  cached = global.mongoose = { conn: null, promise: null };
}

export async function connectDB(): Promise<typeof mongoose> {
  if (cached.conn) return cached.conn;
  if (!cached.promise) {
    cached.promise = mongoose.connect(MONGODB_URI, {
      bufferCommands: false,
      maxPoolSize: 10,
    });
  }
  cached.conn = await cached.promise;
  return cached.conn;
}
```
